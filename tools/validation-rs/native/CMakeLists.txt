cmake_minimum_required(VERSION 3.24)
project(ElectroBowValidationHost VERSION 0.1.0 LANGUAGES C CXX)
if(NOT IS_DIRECTORY "${JUCE_PATH}/modules/juce_audio_processors_headless")
    message(FATAL_ERROR "Set JUCE_PATH to the JUCE checkout used to build the plugin")
endif()
add_subdirectory("${JUCE_PATH}" juce EXCLUDE_FROM_ALL)
juce_add_console_app(validation-host PRODUCT_NAME "validation-host")
target_sources(validation-host PRIVATE vst3.cpp)
target_compile_features(validation-host PRIVATE cxx_std_20)
target_compile_definitions(validation-host PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(validation-host PRIVATE juce::juce_audio_processors_headless)
